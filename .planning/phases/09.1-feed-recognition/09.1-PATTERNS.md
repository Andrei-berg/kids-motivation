```typescript
export interface RankEntry {
  id: string
  name: string
  coinsWeek: number
  avatarUrl?: string | null
  [key: string]: unknown
}

export interface RankResult<T extends RankEntry> {
  podium: T[]
  list: T[]
}

/**
 * Sorts entries descending by coinsWeek (stable tiebreak on id ascending so
 * output is deterministic across calls), then splits ...
 */
export function rankChildren<T extends RankEntry>(entries: T[]): RankResult<T> {
  const sorted = [...entries].sort((a, b) => {
    if (b.coinsWeek !== a.coinsWeek) return b.coinsWeek - a.coinsWeek
    return a.id < b.id ? -1 : a.id > b.id ? 1 : 0
  })
  return { podium: sorted.slice(0, 3), list: sorted.slice(3) }
}
```

**Adapted shape for `pickHighlight`** (deterministic tiebreak matters here too — if two events of
the same rank tier exist for one child today, prefer the same tiebreak style used above, e.g.
most-recent `created_at` wins, documented in a doc-comment the same way):
```typescript
// lib/kid/feed-highlights.ts
// Pure "biggest highlight" picker + rail-color lookup for the Family Feed
// story reel (Phase 9.1). No I/O, no Supabase, no React — operates on
// FeedEvent[] already fetched by components/feed/FamilyFeed.tsx.

import type { FeedEvent, FeedEventKind } from '@/lib/models/feed.types'
import { K } from '@/components/kid/design/kidTheme'

// Highest wins. Achievement-type events outrank the routine `day_filled`
// event (D-04 in .planning/phases/09.1-feed-recognition/09.1-CONTEXT.md).
// `kind === 'boost'` is treated as a single tier — see Pitfall 2 in
// 09.1-RESEARCH.md (weekly_boost/boost_milestone are NOT distinguishable at
// the `kind` level, only via ref_id substring, which this ranking ignores).
export const RANK_ORDER: FeedEventKind[] = [
  'badge', 'medal', 'boost', 'streak',
  'reading_approved', 'reward_approved', 'purchase', 'day_filled',
]

export function pickHighlight(events: FeedEvent[]): FeedEvent | null {
  if (events.length === 0) return null
  const sorted = [...events].sort((a, b) => {
    const ra = RANK_ORDER.indexOf(a.kind)
    const rb = RANK_ORDER.indexOf(b.kind)
    if (ra !== rb) return ra - rb // lower index = higher rank
    return new Date(b.created_at).getTime() - new Date(a.created_at).getTime() // tiebreak: newest wins
  })
  return sorted[0]
}

// D-08 rail color map — locked mapping, see 09.1-UI-SPEC.md "Rail Color Map".
// Sourced directly from K (kidTheme.ts), NOT the FamilyFeed `palette(variant)`
// function — rail colors are variant-independent (Pitfall 3, 09.1-RESEARCH.md).
export const RAIL_COLOR_MAP: Partial<Record<FeedEventKind, string>> = {
  boost: K.mango,
  streak: K.mango,
  medal: K.berry,
  badge: K.berry,
  day_filled: K.mint,
  reading_approved: K.grape,
  level_up: K.grape,
  purchase: K.sky,
  reward_approved: K.sky,
  note: K.line,
  // coins_earned / coins_spent: no emitter exists — intentionally omitted.
}
```

import { describe, it, expect } from 'vitest'

import { pickHighlight, RAIL_COLOR_MAP, RANK_ORDER } from '@/lib/kid/feed-highlights'
import type { FeedEvent } from '@/lib/models/feed.types'

function event(overrides: Partial<FeedEvent>): FeedEvent {
  return {
    id: 'e1', family_id: 'f1', child_id: 'c1', actor_member_id: null,
    kind: 'day_filled', title: 't', body: null, amount: null, icon: null,
    ref_type: 'day', ref_id: null, metadata: {}, created_at: '2026-09-17T10:00:00.000Z',
    ...overrides,
  }
}

describe('pickHighlight', () => {
  it('returns null for zero events', () => {
    expect(pickHighlight([])).toBeNull()
  })

  it('picks badge over day_filled', () => {
    const badge = event({ id: 'b', kind: 'badge' })
    const filled = event({ id: 'd', kind: 'day_filled' })
    expect(pickHighlight([filled, badge])?.id).toBe('b')
  })

  it('breaks a same-rank tie by newest created_at', () => {
    const older = event({ id: 'o', kind: 'medal', created_at: '2026-09-17T09:00:00.000Z' })
    const newer = event({ id: 'n', kind: 'medal', created_at: '2026-09-17T11:00:00.000Z' })
    expect(pickHighlight([older, newer])?.id).toBe('n')
  })
})

describe('RAIL_COLOR_MAP', () => {
  it('covers every kind with a real emitter', () => {
    for (const kind of RANK_ORDER) {
      expect(RAIL_COLOR_MAP[kind]).toBeTruthy()
    }
  })
})
```

// Source: lib/repositories/feed.repo.ts:141-159 (existing, unchanged — do not modify)
function feedSeenKey(familyId: string) {
  return `feed_seen_${familyId}`
}
export function getFeedSeen(familyId: string): string | null {
  try {
    return window.localStorage.getItem(feedSeenKey(familyId))
  } catch {
    return null
  }
}
export function markFeedSeen(familyId: string, atIso?: string): void {
  try {
    window.localStorage.setItem(feedSeenKey(familyId), atIso ?? new Date().toISOString())
  } catch {
    /* private mode / disabled storage — the badge just won't clear, no crash */
  }
}
```

// D-06: per-viewer, per-child, per-day story-seen tracking. Distinct prefix
// (story_seen_) from feed_seen_ above so the two conventions never collide.
function storySeenKey(familyId: string, childId: string, dateStr: string) {
  return `story_seen_${familyId}_${childId}_${dateStr}`
}
export function getStorySeen(familyId: string, childId: string, dateStr: string): boolean {
  try {
    return window.localStorage.getItem(storySeenKey(familyId, childId, dateStr)) !== null
  } catch {
    return false
  }
}
export function markStorySeen(familyId: string, childId: string, dateStr: string): void {
  try {
    window.localStorage.setItem(storySeenKey(familyId, childId, dateStr), new Date().toISOString())
  } catch {
    /* private mode / disabled storage — ring just won't clear, no crash */
  }
}
